class Game_Interpreter
  def map_event(event_name)
    return unless event_name.respond_to?(:to_s)
    name = event_name.to_s.strip

    get_map_event_id = lambda do 
      $game_map.events.each do |id, ev|
        event = ev.event
        next unless event
        
        event.name =~ /<event\s*:*\s*(.+)>/i
        return id if $1 && $1.downcase == name.downcase 
        
        event.pages[0].list.each do |cmd|
          if cmd.code == 108 || cmd.code == 408
            cmd.parameters[0] =~ /<event\s*:*\s*(.+)>/i
            return id if $1 && $1.downcase == name.downcase
          end
        end
      end
      return nil
    end

    map_event_id = get_map_event_id.call
    return unless map_event_id

    map = load_data(sprintf("Data/Map%03d.rvdata2", @map_id))
    event = map.events[map_event_id].pages[0]

    child = Game_Interpreter.new(@depth + 1)
    child.setup(event.list, @event_id)
    child.run
  end
end
